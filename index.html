function crearPDFBlob(datos){ if(!datos.length)throw new Error('No hay datos para exportar.'); if(!window.jspdf||!window.jspdf.jsPDF)throw new Error('No se pudo cargar la librería PDF.'); const {jsPDF}=window.jspdf; const doc=new jsPDF({orientation:'landscape',unit:'mm',format:'a4'}); const pageWidth=doc.internal.pageSize.getWidth(); const pageHeight=doc.internal.pageSize.getHeight(); const margin=10; const cols=[ {title:'Foto',width:34,align:'center'}, {title:'Producto',width:58,align:'left'}, {title:'Inicial',width:25,align:'center'}, {title:'Entregada',width:30,align:'center'}, {title:'Restante',width:28,align:'center'}, {title:'Precio',width:28,align:'center'}, {title:'Total',width:30,align:'center'}, {title:'Valor restante',width:38,align:'center'} ]; const tableWidth=cols.reduce((s,c)=>s+c.width,0); const x0=Math.max(margin,(pageWidth-tableWidth)/2); const headerH=9; let y=28; function dibujarEncabezado(){ doc.setFont('helvetica','bold'); doc.setFontSize(20); doc.setTextColor(6,59,97); doc.text('Control Diario de Reparto',margin,15); doc.setFont('helvetica','normal'); doc.setFontSize(8); doc.setTextColor(83,107,122); doc.text(new Date().toLocaleString('es-SV'),margin,21); let x=x0; cols.forEach(c=>{ doc.setFillColor(6,59,97); doc.setDrawColor(6,59,97); doc.rect(x,y,c.width,headerH,'F'); doc.setTextColor(255,255,255); doc.setFont('helvetica','bold'); doc.setFontSize(8); doc.text(c.title,x+c.width/2,y+6,{align:'center'}); x+=c.width; }); y+=headerH; } function nuevaPagina(){ doc.addPage(); y=28; dibujarEncabezado(); } dibujarEncabezado(); doc.setFont('helvetica','normal'); doc.setFontSize(8); doc.setTextColor(22,54,75); datos.forEach(p=>{ const producto=String(p.Producto??''); const imagen=imagenesProductos.get(p) || (datos.length===1 ? imagenBase64 : ''); const lineas=doc.splitTextToSize(producto,cols[1].width-4); let fotoW=0; let fotoH=0; /* * SOLO AJUSTE DE FOTO: * Se conserva la orientación original de la imagen * y se calcula su tamaño proporcionalmente. */ if(imagen){ try{ const prop=doc.getImageProperties(imagen); const anchoOriginal=Number(prop.width)||1; const altoOriginal=Number(prop.height)||1; let maxW; let maxH; // Foto horizontal if(anchoOriginal>=altoOriginal){ maxW=29; maxH=22; } // Foto vertical else{ maxW=22; maxH=29; } const escala=Math.min( maxW/anchoOriginal, maxH/altoOriginal ); fotoW=anchoOriginal*escala; fotoH=altoOriginal*escala; }catch(e){ fotoW=0; fotoH=0; } } /* * La fila se adapta al tamaño de la foto. * Si no hay foto, conserva una altura normal. */ const altoFoto=imagen&&fotoH ? fotoH+8 : 9; const alto=Math.max( 9, lineas.length*4+5, altoFoto ); if(y+alto>pageHeight-10){ nuevaPagina(); } let x=x0; cols.forEach((c,i)=>{ doc.setFillColor(255,255,255); doc.setDrawColor(201,216,223); doc.rect(x,y,c.width,alto,'FD'); if(i===0){ if(imagen&&fotoW&&fotoH){ try{ const prop=doc.getImageProperties(imagen); /* * Centrado exacto dentro de la celda. * Se mantiene la proporción de la imagen. */ const xi=x+(c.width-fotoW)/2; const yi=y+(alto-fotoH)/2; doc.addImage( imagen, prop.fileType, xi, yi, fotoW, fotoH, undefined, 'FAST' ); }catch(e){} } }else{ let valor=''; if(i===1)valor=producto; if(i===2)valor=String(p.CantidadInicial??0); if(i===3)valor=String(p.CantidadEntregada??0); if(i===4)valor=String(p.CantidadRestante??0); if(i===5)valor=dinero(p.Precio); if(i===6)valor=dinero(p.Total); if(i===7)valor=dinero(p.ValorRestante); const texto=i===1?lineas:[valor]; const tx=c.align==='left' ?x+2 :x+c.width/2; const opts=c.align==='left' ?{} :{align:'center'}; doc.text( texto, tx, y+5, opts ); } x+=c.width; }); y+=alto; }); return doc.output('blob'); }
